2 Copiar el SQL

Haz clic en "Copiar SQL" para copiar la migración completa:

-- ===================================================== -- FIX RLS POLICIES FOR AUDIO STORAGE AND CHECKPOINTS -- ===================================================== -- PARTE 1: ARREGLAR STORAGE BUCKET (consultation-audio) INSERT INTO storage.buckets (id, name, public) VALUES ('consultation-audio', 'consultation-audio', false) ON CONFLICT (id) DO UPDATE SET public = false; DROP POLICY IF EXISTS "Authenticated users can upload audio" ON storage.objects; DROP POLICY IF EXISTS "Authenticated users can read audio" ON storage.objects; DROP POLICY IF EXISTS "Authenticated users can update audio" ON storage.objects; DROP POLICY IF EXISTS "Authenticated users can delete audio" ON storage.objects; CREATE POLICY "Authenticated users can upload audio" ON storage.objects FOR INSERT TO authenticated WITH CHECK ( bucket_id = 'consultation-audio' ); CREATE POLICY "Authenticated users can read audio" ON storage.objects FOR SELECT TO authenticated USING ( bucket_id = 'consultation-audio' ); CREATE POLICY "Authenticated users can update audio" ON storage.objects FOR UPDATE TO authenticated USING (bucket_id = 'consultation-audio') WITH CHECK (bucket_id = 'consultation-audio'); CREATE POLICY "Authenticated users can delete audio" ON storage.objects FOR DELETE TO authenticated USING (bucket_id = 'consultation-audio'); -- PARTE 2: ARREGLAR SESSION_CHECKPOINTS TABLE ALTER TABLE session_checkpoints ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Users can select own checkpoints" ON session_checkpoints; DROP POLICY IF EXISTS "Users can insert own checkpoints" ON session_checkpoints; DROP POLICY IF EXISTS "Users can update own checkpoints" ON session_checkpoints; DROP POLICY IF EXISTS "Users can delete own checkpoints" ON session_checkpoints; CREATE POLICY "Users can select own checkpoints" ON session_checkpoints FOR SELECT TO authenticated USING (auth.uid() = user_id); CREATE POLICY "Users can insert own checkpoints" ON session_checkpoints FOR INSERT TO authenticated WITH CHECK (auth.uid() = user_id); CREATE POLICY "Users can update own checkpoints" ON session_checkpoints FOR UPDATE TO authenticated USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id); CREATE POLICY "Users can delete own checkpoints" ON session_checkpoints FOR DELETE TO authenticated USING (auth.uid() = user_id);
